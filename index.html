const [selectedId, setSelectedId] = useState(null);
const [search, setSearch] = useState('');
const [showDetailMobile, setShowDetailMobile] = useState(false);
const [editingPriceId, setEditingPriceId] = useState(null);
const [priceDraft, setPriceDraft] = useState({ price: '', taxType: 'included', memo: '' });
const [confirmAction, setConfirmAction] = useState(null); // { type: 'archive'|'restore'|'delete', id }
const [archiveForm, setArchiveForm] = useState({ date: '', reason: '' });
const [showSearchOverlay, setShowSearchOverlay] = useState(false);
const [globalSearch, setGlobalSearch] = useState('');
const [draft, setDraft] = useState(null);
const [newHistoryEntry, setNewHistoryEntry] = useState({ date: todayStr(), content: '' });
const [paymentMethods, setPaymentMethods] = useState(['銀行振込', '口座振替', '現金', '掛売（月末請求）']);
const [paymentMethodsLoaded, setPaymentMethodsLoaded] = useState(false);
const [showAddPaymentMethod, setShowAddPaymentMethod] = useState(false);
const [newPaymentMethod, setNewPaymentMethod] = useState('');
const paymentSaveTimer = useRef(null);
const saveTimer = useRef(null);

const skipNextCustomersSave = useRef(false);
const skipNextPaymentSave = useRef(false);
const [syncError, setSyncError] = useState(null);

// 顧客データ：Firestoreの1つのドキュメントを全員で共有し、変更をリアルタイムに反映
useEffect(() => {
const ref = window.db.collection('app').doc('customers');
const unsub = ref.onSnapshot(
(snap) => {
if (snap.exists) {
const data = snap.data();
skipNextCustomersSave.current = true;
setCustomers(Array.isArray(data.list) ? data.list : []);
} else {
const initial = [sampleCustomer()];
ref.set({ list: initial }).catch((e) => setSyncError(String(e)));
}
setLoaded(true);
},
(e) => setSyncError(String(e))
);
return unsub;
}, []);

useEffect(() => {
if (!loaded) return;
if (skipNextCustomersSave.current) {
skipNextCustomersSave.current = false;
return;
}
clearTimeout(saveTimer.current);
saveTimer.current = setTimeout(() => {
const ref = window.db.collection('app').doc('customers');
ref.set({ list: customers }).catch((e) => setSyncError(String(e)));
}, 400);
return () => clearTimeout(saveTimer.current);
}, [customers, loaded]);

// 支払方法一覧：こちらもFirestoreの1つのドキュメントで全員共有
useEffect(() => {
const ref = window.db.collection('app').doc('paymentMethods');
const unsub = ref.onSnapshot(
(snap) => {
if (snap.exists) {
const data = snap.data();
if (Array.isArray(data.list) && data.list.length) {
skipNextPaymentSave.current = true;
setPaymentMethods(data.list);
}
} else {
ref.set({ list: paymentMethods }).catch((e) => setSyncError(String(e)));
}
setPaymentMethodsLoaded(true);
},
(e) => setSyncError(String(e))
);
return unsub;
// eslint-disable-next-line react-hooks/exhaustive-deps
}, []);

useEffect(() => {
if (!paymentMethodsLoaded) return;
if (skipNextPaymentSave.current) {
skipNextPaymentSave.current = false;
return;
}
clearTimeout(paymentSaveTimer.current);
paymentSaveTimer.current = setTimeout(() => {
const ref = window.db.collection('app').doc('paymentMethods');
ref.set({ list: paymentMethods }).catch((e) => setSyncError(String(e)));
}, 400);
return () => clearTimeout(paymentSaveTimer.current);
}, [paymentMethods, paymentMethodsLoaded]);

const selected = customers.find((c) => c.id === selectedId) || null;

useEffect(() => {
const c = customers.find((c) => c.id === selectedId);
setDraft(c ? basicFields(c) : null);
// eslint-disable-next-line react-hooks/exhaustive-deps
}, [selectedId]);

const dv = draft || (selected ? basicFields(selected) : {});
const setDraftField = (field, value) => setDraft({ ...dv, [field]: value });
const isDirty = !!(selected && JSON.stringify(dv) !== JSON.stringify(basicFields(selected)));
const saveBasicInfo = () => {
if (!selected) return;
updateCustomer(selected.id, { ...dv });
setDraft(null);
};
const openInGoogleMaps = () => {
const q = (dv.address || '').trim();
if (!q) return;
window.open(`https://www.google.com/maps/search/?api=1&query=${encodeURIComponent(q)}`, '_blank');
};

const addPaymentMethod = () => {
const m = newPaymentMethod.trim();
if (!m) return;
setPaymentMethods((prev) => (prev.includes(m) ? prev : [...prev, m]));
setDraftField('paymentMethod', m);
setNewPaymentMethod('');
setShowAddPaymentMethod(false);
};

const updateCustomer = (id, patch) => {
setCustomers((prev) =>
prev.map((c) => (c.id === id ? { ...c, ...(typeof patch === 'function' ? patch(c) : patch) } : c))
);
};

const list = customers
.filter((c) => c.status === view)
.filter((c) => {
if (!search.trim()) return true;
const q = search.trim();
return (c.companyName || '').includes(q) || (c.contactName || '').includes(q);
})
.sort((a, b) => (a.companyName || '').localeCompare(b.companyName || '', 'ja'));

const activeCount = customers.filter((c) => c.status === 'active').length;
const archivedCount = customers.filter((c) => c.status === 'archived').length;

const globalResults = globalSearch.trim()
? customers
.filter((c) => (c.companyName || '').includes(globalSearch.trim()))
.sort((a, b) => (a.companyName || '').localeCompare(b.companyName || '', 'ja'))
: [];

const openFromSearch = (c) => {
setView(c.status);
setSelectedId(c.id);
setShowDetailMobile(true);
setShowSearchOverlay(false);
setGlobalSearch('');
};

const backToList = () => {
setSelectedId(null);
setShowDetailMobile(false);
};

const handleAddNew = () => {
const nc = emptyCustomer();
setCustomers((prev) => [nc, ...prev]);
setSelectedId(nc.id);
setView('active');
setShowDetailMobile(true);
};

const handleSelect = (id) => {
setSelectedId(id);
setShowDetailMobile(true);
setEditingPriceId(null);
};

const addMenuItem = (customerId) => {
updateCustomer(customerId, (c) => ({
menuItems: [...c.menuItems, { id: uid(), name: '', price: 0, taxType: 'included' }],
}));
};

const removeMenuItem = (customerId, itemId) => {
updateCustomer(customerId, (c) => ({
menuItems: c.menuItems.filter((mi) => mi.id !== itemId),
}));
};

const renameMenuItem = (customerId, itemId, name) => {
updateCustomer(customerId, (c) => ({
menuItems: c.menuItems.map((mi) => (mi.id === itemId ? { ...mi, name } : mi)),
}));
};

const addManualHistoryEntry = (customerId) => {
const content = newHistoryEntry.content.trim();
if (!content) return;
updateCustomer(customerId, (c) => ({
historyLog: [
{ id: uid(), date: newHistoryEntry.date || todayStr(), content },
...c.historyLog,
],
}));
setNewHistoryEntry({ date: todayStr(), content: '' });
};

const startPriceEdit = (item) => {
setEditingPriceId(item.id);
setPriceDraft({ price: String(item.price ?? 0), taxType: item.taxType || 'included', memo: '' });
};

const confirmPriceEdit = (customerId, item) => {
const newPrice = Number(priceDraft.price) || 0;
updateCustomer(customerId, (c) => {
const menuItems = c.menuItems.map((mi) =>
mi.id === item.id ? { ...mi, price: newPrice, taxType: priceDraft.taxType } : mi
);
let historyLog = c.historyLog;
if (newPrice !== item.price) {
historyLog = [
{
id: uid(),
date: todayStr(),
itemName: item.name || '(名称未設定)',
oldPrice: item.price,
newPrice,
memo: priceDraft.memo,
},
...c.historyLog,
];
}
return { menuItems, historyLog };
});
setEditingPriceId(null);
};

const runConfirmedAction = () => {
if (!confirmAction) return;
const { type, id } = confirmAction;
if (type === 'archive') {
updateCustomer(id, {
status: 'archived',
archivedAt: archiveForm.date || todayStr(),
endReason: archiveForm.reason,
});
setShowDetailMobile(false);
} else if (type === 'restore') {
updateCustomer(id, { status: 'active', archivedAt: null, endReason: '' });
} else if (type === 'delete') {
setCustomers((prev) => prev.filter((c) => c.id !== id));
setSelectedId(null);
setShowDetailMobile(false);
}
setConfirmAction(null);
};

return (
<div className="cms-app">
{syncError && (
<div className="cms-sync-error">
データベースに接続できません。firebaseConfig の設定をご確認ください。({syncError})
</div>
)}

<div className="cms-topbar">
<div className="cms-topbar__title">
<h1 className="cms-display">株式会社料理かしば</h1>
<p>顧客取引データベース</p>
</div>
<div style={{ display: 'flex', gap: 8, flexWrap: 'wrap' }}>
<button className="cms-btn cms-btn--ghost-dark" onClick={() => setShowSearchOverlay(true)}>
<Icon name="search" size={15} /> 会社名で検索
</button>
<button className="cms-btn cms-btn--accent" onClick={handleAddNew}>
<Icon name="plus" size={15} /> 新規取引先を登録
</button>
</div>
</div>

<div className="cms-body cms-relative">
<div
className="cms-list-pane"
style={{ display: showDetailMobile ? 'none' : 'flex' }}
>
<div className="cms-tabs">
<button
className={`cms-tab ${view === 'active' ? 'cms-tab--active' : ''}`}
onClick={() => setView('active')}
>
取引先一覧 <span className="cms-tab__count">({activeCount})</span>
</button>
<button
className={`cms-tab ${view === 'archived' ? 'cms-tab--active' : ''}`}
onClick={() => setView('archived')}
>
取引終了BOX <span className="cms-tab__count">({archivedCount})</span>
</button>
</div>

<div className="cms-search">
<Icon name="search" size={14} />
<input
type="text"
placeholder="会社名・担当者名で検索"
value={search}
onChange={(e) => setSearch(e.target.value)}
/>
</div>

<div className="cms-list">
{list.length === 0 && (
<div className="cms-list-empty">
{view === 'active' ? '取引先が登録されていません。' : '取引終了した取引先はありません。'}
</div>
)}
{list.map((c) => (
<div
key={c.id}
className={`cms-list-item ${c.id === selectedId ? 'cms-list-item--selected' : ''}`}
onClick={() => handleSelect(c.id)}
>
<p className="cms-list-item__name">{c.companyName || '（会社名未入力）'}</p>
<p className="cms-list-item__sub">
<Icon name="user" size={11} /> {c.contactName || '担当者未設定'}
<span>・</span>
<Icon name="package" size={11} /> {c.menuItems.length}品目
{c.status === 'archived' && c.archivedAt && (
<>
<span>・</span>
終了 {c.archivedAt}
</>
)}
</p>
</div>
))}
</div>
</div>

<div
className="cms-detail-pane"
style={{ display: showDetailMobile ? 'block' : undefined }}
>
{!selected && (
<div className="cms-detail-empty">
<Icon name="building" size={40} />
<p>左の一覧から取引先を選択するか、<br />「新規取引先を登録」から追加してください。</p>
</div>
)}

{selected && (
<>
<div className="cms-detail-header">
<div className="cms-detail-header__top">
<button
className="cms-btn cms-btn--ghost cms-btn--sm cms-back-btn"
onClick={backToList}
>
<Icon name="chevronleft" size={14} /> 取引先一覧へ戻る
</button>
<input
className="cms-company-name"
placeholder="会社名を入力"
value={dv.companyName || ''}
onChange={(e) => setDraftField('companyName', e.target.value)}
/>
</div>
<div className="cms-detail-header__actions">
<button
className="cms-btn cms-btn--accent cms-btn--sm"
onClick={saveBasicInfo}
disabled={!isDirty}
>
<Icon name="save" size={13} /> 修正を保存
</button>
{selected.status === 'active' ? (
<button
className="cms-btn cms-btn--ghost cms-btn--sm"
onClick={() => {
setArchiveForm({ date: todayStr(), reason: '' });
setConfirmAction({ type: 'archive', id: selected.id });
}}
>
<Icon name="archive" size={13} /> 取引終了にする
</button>
) : (
<button
className="cms-btn cms-btn--ghost cms-btn--sm"
onClick={() => setConfirmAction({ type: 'restore', id: selected.id })}
>
<Icon name="archiverestore" size={13} /> 取引再開する
</button>
)}
<button
className="cms-btn cms-btn--danger cms-btn--sm"
onClick={() => setConfirmAction({ type: 'delete', id: selected.id })}
>
<Icon name="trash" size={13} />
</button>
</div>
</div>

{selected.status === 'archived' && (
<div className="cms-endinfo">
<div className="cms-endinfo__row">
<span className="cms-endinfo__label">終了日</span>
<input
type="date"
value={selected.archivedAt || ''}
onChange={(e) => updateCustomer(selected.id, { archivedAt: e.target.value })}
/>
</div>
<div className="cms-endinfo__row">
<span className="cms-endinfo__label">終了理由</span>
<textarea
placeholder="終了理由を入力"
value={selected.endReason}
onChange={(e) => updateCustomer(selected.id, { endReason: e.target.value })}
/>
</div>
</div>
)}

<div className="cms-section">
<h3 className="cms-section__title"><Icon name="building" size={15} /> 基本情報 <span className="cms-rule" /></h3>
<div className="cms-grid">
<Field label="取引先コード">
<input
type="text"
placeholder="C-0001"
value={dv.customerCode || ''}
onChange={(e) => setDraftField('customerCode', e.target.value)}
/>
</Field>
<Field label="郵便番号" icon="mappin">
<input
type="text"
placeholder="750-0000"
value={dv.postalCode || ''}
onChange={(e) => setDraftField('postalCode', e.target.value)}
/>
</Field>
<Field label="電話番号" icon="phone">
<input
type="text"
placeholder="083-000-0000"
value={dv.tel || ''}
onChange={(e) => setDraftField('tel', e.target.value)}
/>
</Field>
<div className="cms-grid--full" style={{ gridColumn: '1 / -1' }}>
<Field label="住所" icon="mappin">
<div className="cms-address-row">
<input
type="text"
placeholder="山口県下関市〇〇町1-2-3"
value={dv.address || ''}
onChange={(e) => setDraftField('address', e.target.value)}
/>
<button
type="button"
className="cms-btn cms-btn--ghost cms-btn--sm"
onClick={openInGoogleMaps}
disabled={!(dv.address || '').trim()}
>
<Icon name="external" size={13} /> Googleマップで開く
</button>
</div>
</Field>
</div>
<Field label="FAX">
<input
type="text"
value={dv.fax || ''}
onChange={(e) => setDraftField('fax', e.target.value)}
/>
</Field>
<Field label="メールアドレス" icon="mail">
<input
type="text"
value={dv.email || ''}
onChange={(e) => setDraftField('email', e.target.value)}
/>
</Field>
<Field label="担当者名" icon="user">
<input
type="text"
value={dv.contactName || ''}
onChange={(e) => setDraftField('contactName', e.target.value)}
/>
</Field>
<Field label="部署・役職">
<input
type="text"
value={dv.contactRole || ''}
onChange={(e) => setDraftField('contactRole', e.target.value)}
/>
</Field>
<Field label="おおよその食数">
<input
type="text"
placeholder="例：約50食/日"
value={dv.mealCount || ''}
onChange={(e) => setDraftField('mealCount', e.target.value)}
/>
</Field>
<Field label="支払の締め日">
<input
type="text"
placeholder="例：毎月20日締め・翌月末払い"
value={dv.paymentDueDay || ''}
onChange={(e) => setDraftField('paymentDueDay', e.target.value)}
/>
</Field>
<div className="cms-grid--full" style={{ gridColumn: '1 / -1' }}>
<Field label="支払方法">
<div className="cms-payment-row">
<select
value={dv.paymentMethod || ''}
onChange={(e) => setDraftField('paymentMethod', e.target.value)}
>
<option value="">未選択</option>
{paymentMethods.map((m) => (
<option key={m} value={m}>{m}</option>
))}
</select>
<button
type="button"
className="cms-btn cms-btn--ghost cms-btn--sm"
onClick={() => setShowAddPaymentMethod((v) => !v)}
>
<Icon name="plus" size={13} /> 支払方法を追加
</button>
</div>
{showAddPaymentMethod && (
<div className="cms-payment-add">
<input
type="text"
placeholder="新しい支払方法を入力（例：クレジットカード）"
value={newPaymentMethod}
onChange={(e) => setNewPaymentMethod(e.target.value)}
onKeyDown={(e) => e.key === 'Enter' && addPaymentMethod()}
autoFocus
/>
<button className="cms-icon-btn" onClick={addPaymentMethod} aria-label="追加を確定">
<Icon name="check" size={16} />
</button>
<button
className="cms-icon-btn"
onClick={() => { setShowAddPaymentMethod(false); setNewPaymentMethod(''); }}
aria-label="キャンセル"
>
<Icon name="x" size={16} />
</button>
</div>
)}
</Field>
</div>
</div>
</div>

<div className="cms-section">
<h3 className="cms-section__title"><Icon name="package" size={15} /> 注文メニュー <span className="cms-rule" /></h3>

{selected.menuItems.length === 0 && (
<p className="cms-empty-menu">メニューが登録されていません。「＋ メニューを追加」から登録してください。</p>
)}

{selected.menuItems.map((item) =>
editingPriceId === item.id ? (
<div className="cms-price-editor" key={item.id}>
<span style={{ fontSize: 13, fontWeight: 500 }}>{item.name || '（名称未設定）'}</span>
<input
type="number"
data-role="price"
value={priceDraft.price}
onChange={(e) => setPriceDraft((d) => ({ ...d, price: e.target.value }))}
/>
<select
value={priceDraft.taxType}
onChange={(e) => setPriceDraft((d) => ({ ...d, taxType: e.target.value }))}
>
<option value="included">内税</option>
<option value="excluded">外税</option>
</select>
<input
type="text"
data-role="memo"
placeholder="変更メモ（任意・例：原材料高騰のため）"
value={priceDraft.memo}
onChange={(e) => setPriceDraft((d) => ({ ...d, memo: e.target.value }))}
/>
<button className="cms-icon-btn" onClick={() => confirmPriceEdit(selected.id, item)} aria-label="保存">
<Icon name="check" size={16} />
</button>
<button className="cms-icon-btn" onClick={() => setEditingPriceId(null)} aria-label="キャンセル">
<Icon name="x" size={16} />
</button>
</div>
) : (
<div className="cms-menu-item" key={item.id}>
<input
type="text"
placeholder="メニュー名（例：上、特上、特別メニュー）"
value={item.name}
onChange={(e) => renameMenuItem(selected.id, item.id, e.target.value)}
/>
<span className={`cms-tax-badge cms-tax-badge--${item.taxType}`}>{TAX_LABEL[item.taxType]}</span>
<span className="cms-menu-item__price">{yen(item.price)}</span>
<button className="cms-icon-btn" onClick={() => startPriceEdit(item)} aria-label="価格を編集">
<Icon name="pen" size={15} />
</button>
<button
className="cms-icon-btn cms-icon-btn--danger"
onClick={() => removeMenuItem(selected.id, item.id)}
aria-label="メニューを削除"
>
<Icon name="trash" size={15} />
</button>
</div>
)
)}

<button className="cms-btn cms-btn--ghost cms-btn--sm" onClick={() => addMenuItem(selected.id)}>
<Icon name="plus" size={13} /> メニューを追加
</button>
</div>

<div className="cms-section">
<h3 className="cms-section__title"><Icon name="stickynote" size={15} /> 備考・注意事項 <span className="cms-rule" /></h3>
<Field label="">
<textarea
placeholder="配送に関する注意事項、アレルギー対応、締切時間など"
value={dv.notes || ''}
onChange={(e) => setDraftField('notes', e.target.value)}
/>
</Field>
</div>

<div className="cms-section" style={{ borderBottom: 'none' }}>
<h3 className="cms-section__title"><Icon name="history" size={15} /> 取引先変更履歴 <span className="cms-rule" /></h3>

<div className="cms-history-add">
<input
type="date"
value={newHistoryEntry.date}
onChange={(e) => setNewHistoryEntry((f) => ({ ...f, date: e.target.value }))}
/>
<input
type="text"
placeholder="変更内容を記入（例：担当者が佐藤様に変更、配送時間を9時に変更）"
value={newHistoryEntry.content}
onChange={(e) => setNewHistoryEntry((f) => ({ ...f, content: e.target.value }))}
/>
<button
className="cms-btn cms-btn--ghost cms-btn--sm"
onClick={() => addManualHistoryEntry(selected.id)}
disabled={!newHistoryEntry.content.trim()}
>
<Icon name="plus" size={13} /> 追加
</button>
</div>

{selected.historyLog.length === 0 ? (
<p className="cms-no-history">変更履歴の記録はまだありません。価格を編集すると自動で記録されるほか、上の欄から自由に記入できます。</p>
) : (
selected.historyLog.map((h) => (
<div className="cms-history-item" key={h.id}>
<span className="cms-history-date">{h.date}</span>
<div className="cms-history-detail">
{h.itemName ? (
<>
<div>
<strong>{h.itemName}</strong>{' '}
<span className="cms-price-change">
{yen(h.oldPrice)} → <b>{yen(h.newPrice)}</b>
</span>
</div>
{h.memo && <div className="cms-history-memo">{h.memo}</div>}
</>
) : (
<div>{h.content}</div>
)}
</div>
</div>
))
)}
</div>
</>
)}
</div>

{showSearchOverlay && (
<div className="cms-confirm-overlay" onClick={() => setShowSearchOverlay(false)}>
<div className="cms-search-box" onClick={(e) => e.stopPropagation()}>
<div className="cms-search-box__head">
<Icon name="search" size={15} />
<input
type="text"
autoFocus
placeholder="会社名を入力して検索（取引終了BOXも対象）"
value={globalSearch}
onChange={(e) => setGlobalSearch(e.target.value)}
/>
<button className="cms-icon-btn" onClick={() => setShowSearchOverlay(false)} aria-label="閉じる">
<Icon name="x" size={18} />
</button>
</div>
<div className="cms-search-box__results">
{!globalSearch.trim() && (
<p className="cms-no-history">会社名を入力すると、取引先一覧・取引終了BOXの両方から検索します。</p>
)}
{globalSearch.trim() && globalResults.length === 0 && (
<p className="cms-no-history">該当する会社が見つかりませんでした。</p>
)}
{globalResults.map((c) => (
<div key={c.id} className="cms-search-result" onClick={() => openFromSearch(c)}>
<div>
<p className="cms-list-item__name">{c.companyName || '（会社名未入力）'}</p>
<p className="cms-list-item__sub">{c.address || '住所未入力'}</p>
</div>
<span className={`cms-tax-badge ${c.status === 'active' ? 'cms-tax-badge--included' : 'cms-tax-badge--excluded'}`}>
{c.status === 'active' ? '取引中' : '終了'}
</span>
</div>
))}
</div>
</div>
</div>
)}

{confirmAction && (
<div className="cms-confirm-overlay">
<div className="cms-confirm-box">
{confirmAction.type === 'archive' && (
<>
<p style={{ marginBottom: 12 }}>この取引先を「取引終了BOX」に移動します。終了日と終了理由を入力してください。</p>
<div className="cms-field" style={{ marginBottom: 10 }}>
<span className="cms-field__label">終了日</span>
<input
type="date"
value={archiveForm.date}
onChange={(e) => setArchiveForm((f) => ({ ...f, date: e.target.value }))}
/>
</div>
<div className="cms-field" style={{ marginBottom: 16 }}>
<span className="cms-field__label">終了理由（任意）</span>
<textarea
placeholder="例：先方廃業のため、価格折り合わず、取扱店舗変更のため"
value={archiveForm.reason}
onChange={(e) => setArchiveForm((f) => ({ ...f, reason: e.target.value }))}
style={{ minHeight: 60 }}
/>
</div>
</>
)}
{confirmAction.type === 'restore' && (
<p>この取引先を取引先一覧に戻します。よろしいですか？</p>
)}
{confirmAction.type === 'delete' && (
<p>
<Icon name="alert" size={14} style={{ verticalAlign: '-2px', marginRight: 4, color: 'var(--accent-ink)' }} />
このデータを完全に削除します。この操作は取り消せません。よろしいですか？
</p>
)}
<div className="cms-confirm-box__actions">
<button className="cms-btn cms-btn--ghost cms-btn--sm" onClick={() => setConfirmAction(null)}>
キャンセル
</button>
<button
className={`cms-btn cms-btn--sm ${confirmAction.type === 'delete' ? 'cms-btn--danger' : 'cms-btn--accent'}`}
onClick={runConfirmedAction}
>
{confirmAction.type === 'archive' && '取引終了にする'}
{confirmAction.type === 'restore' && '取引を再開する'}
{confirmAction.type === 'delete' && '削除する'}
</button>
</div>
</div>
</div>
)}
</div>
</div>
);
}


const root = ReactDOM.createRoot(document.getElementById('root'));
root.render(<CustomerManagementApp />);
